==================
Legacy proc, channel declaration, and spawn
==================

proc Child<N: u32> {
  output: chan<uN[N]> out;
  config(output: chan<uN[N]> out) { (output,) }
  init { uN[N]:0 }
  next(state: uN[N]) {
    let tok = send(join(), output, state);
    state + uN[N]:1
  }
}

#[test_proc]
proc Network {
  output: chan<u8> out;
  config() {
    let (sender, receiver) = chan<u8>("network");
    spawn Child<u32:8>(sender);
    (receiver,)
  }
  init { u8:0 }
  next(state: u8) { state }
}

==================
Impl-style proc and proc alias
==================

proc Counter<N: u32> { output: chan<uN[N]> out, amount: uN[N], }
proc ByteCounter = Counter<u32:8>;

impl Counter<u32:8> {
  fn config(output: chan<u8> out, amount: u8) -> (Self,) { (Self { output, amount },) }
  fn init() -> u8 { u8:0 }
  fn next(self: Self, state: u8) -> u8 {
    let tok = send(join(), self.output, state);
    state + self.amount
  }
}
