fn classify(x: u8, pair: (u8, u8)) -> u8 {
  let (_, y) = pair;
  match x {
    u8:0 | u8:1 => y,
    u8:2..=u8:7 => u8:2,
    _ => u8:3,
  }
}

fn loops<N: u32>(xs: u32[N]) -> u32 {
  let sum = for (i, acc): (u32, u32) in u32:0..N {
    acc + xs[i]
  } (u32:0);
  unroll_for! (i, acc) in u32:0..N { acc + xs[i] } (sum)
}
