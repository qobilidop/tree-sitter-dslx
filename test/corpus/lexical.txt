// A module comment.
fn lexical<N': u32>(value': uN[N']) -> uN[N'] {
  'loop: for (i, acc) in u32:0..N' {
    // Comments are extras inside expressions.
    acc + value'
  } (uN[N']:0)
}

fn escapes() {
  let quote = '\'';
  let hex = '\x41';
  let text = "tab:\t slash:\\ unicode:\u{3bb}";
  (quote, hex, text)
}
