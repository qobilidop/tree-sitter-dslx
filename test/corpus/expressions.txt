fn operators(a: u32, b: u32, c: u32) -> bool {

  let a = x[u32:1];
  let b = x[:u32:2];
  let c = x[u32:1:];
  let d = x[u32:1+:u8];

  let f = s.x;
  (a, b, c, d, e, f)
